// Three.js r186 - Node System

// directives


// structs


// uniforms
@binding( 2 ) @group( 1 ) var nodeUniform3 : texture_2d<f32>;
@binding( 3 ) @group( 1 ) var nodeUniform8 : texture_2d<f32>;

struct renderStruct {
	cameraNear : f32,
	cameraFar : f32,
	cameraProjectionMatrix : mat4x4<f32>,
	nodeUniform16 : vec2<f32>,
	nodeUniform17 : vec2<f32>,
	cameraViewMatrix : mat4x4<f32>
};
@binding( 0 ) @group( 0 )
var<uniform> render : renderStruct;

struct objectStruct {
	nodeUniform1 : mat4x4<f32>,
	nodeUniform2 : f32,
	nodeUniform4 : f32,
	nodeUniform9 : vec3<f32>,
	nodeUniform10 : vec3<f32>,
	nodeUniform11 : f32,
	nodeUniform12 : vec3<f32>,
	nodeUniform13 : vec3<f32>,
	nodeUniform14 : f32,
	nodeUniform18 : vec3<f32>,
	nodeUniform19 : f32,
	nodeUniform20 : vec3<f32>,
	nodeUniform21 : f32,
	nodeUniform22 : f32,
	nodeUniform23 : f32,
	nodeUniform24 : f32
};
@binding( 0 ) @group( 1 )
var<uniform> object : objectStruct;

// varyings

struct VaryingsStruct {
	@location( 0 ) vViewCenter : vec3<f32>,
	@location( 1 ) vUv : vec2<f32>,
	@location( 2 ) vRadius : f32,
	@location( 3 ) vSlot : f32,
	@location( 4 ) vColor : vec3<f32>,
	@location( 5 ) vOcclusion : f32,
	@location( 6 ) vPixelRadius : f32,
	@location( 7 ) vAtomId : f32,
	@builtin( position ) builtinClipSpace : vec4<f32>
};
var<private> varyings : VaryingsStruct;

// vars
var<private> modelViewMatrix : mat4x4<f32>;
var<private> nodeVar0 : f32;
var<private> nodeVar14 : vec4<f32>;
var<private> nodeVar25 : f32;
var<private> nodeVar40 : vec4<f32>;

// codes


@vertex
fn main( @builtin( instance_index ) instanceIndex : u32,
	@location( 0 ) instancePosition : vec3<f32>,
	@location( 1 ) instanceTargetPosition : vec3<f32>,
	@location( 2 ) position : vec3<f32>,
	@location( 3 ) instancePacked : vec4<f32> ) -> VaryingsStruct {

	// flow
	// code

	modelViewMatrix = ( render.cameraViewMatrix * object.nodeUniform1 );
	varyings.vViewCenter = ( modelViewMatrix * vec4<f32>( mix( instancePosition, instanceTargetPosition, object.nodeUniform2 ), 1.0 ) ).xyz;
	varyings.vUv = position.xy;
	nodeVar0 = textureLoad( nodeUniform3, vec2<i32>( i32( round( ( instancePacked.x * 255.0 ) ) ), 0 ), u32( 0u ) ).x;
	varyings.vRadius = nodeVar0;
	varyings.vSlot = f32( i32( round( ( instancePacked.x * 255.0 ) ) ) );
	nodeVar14 = textureLoad( nodeUniform8, vec2<i32>( i32( round( ( instancePacked.x * 255.0 ) ) ), 0 ), u32( 0u ) );
	varyings.vColor = nodeVar14.xyz;
	varyings.vOcclusion = instancePacked.y;

	if ( ( object.nodeUniform14 > 0.5 ) ) {

		nodeVar25 = ( nodeVar0 * ( ( abs( render.cameraProjectionMatrix[ 1u ].y ) * render.nodeUniform16.y ) * 0.5 ) );

	} else {

		nodeVar25 = ( ( nodeVar0 * ( ( abs( render.cameraProjectionMatrix[ 1u ].y ) * render.nodeUniform17.y ) * 0.5 ) ) / max( ( - ( modelViewMatrix * vec4<f32>( mix( instancePosition, instanceTargetPosition, object.nodeUniform2 ), 1.0 ) ).xyz.z ), 0.0001 ) );

	}

	varyings.vPixelRadius = nodeVar25;
	varyings.vAtomId = f32( instanceIndex );

	if ( ( ( nodeVar0 <= 0.0 ) || ( nodeVar25 < object.nodeUniform24 ) ) ) {

		nodeVar40 = vec4<f32>( 2.0, 2.0, 2.0, 1.0 );

	} else {

		nodeVar40 = ( render.cameraProjectionMatrix * vec4<f32>( vec3<f32>( ( ( modelViewMatrix * vec4<f32>( mix( instancePosition, instanceTargetPosition, object.nodeUniform2 ), 1.0 ) ).xyz.xy + ( position.xy * vec2<f32>( ( nodeVar0 * 1.3 ) ) ) ), ( ( modelViewMatrix * vec4<f32>( mix( instancePosition, instanceTargetPosition, object.nodeUniform2 ), 1.0 ) ).xyz.z + nodeVar0 ) ), 1.0 ) );

	}


	// result

	varyings.builtinClipSpace = nodeVar40;

	return varyings;

}
